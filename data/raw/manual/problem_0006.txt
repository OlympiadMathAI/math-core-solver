ID: 0006

SOURCE: IMO Shortlist

YEAR: 2010

TOPIC: Algebra

SUBTOPICS: Functional Equations, Positive Integers

STATEMENT:
Let $f,g:\mathbb{N}\to\mathbb{N}$, where $\mathbb{N}$ is the set of positive integers. Suppose that
\[
f(g(n))=f(n)+1,
\qquad
g(f(n))=g(n)+1
\]
for every positive integer $n$. Prove that
\[
f(n)=g(n)
\]
for every positive integer $n$.

SOLUTION 1:
For a map $h:\mathbb{N}\to\mathbb{N}$, write $h^k$ for its $k$-fold iterate, with $h^0(x)=x$. Repeated use of the hypotheses gives
\[
f(g^k(x))=f(x)+k,
\qquad
g(f^k(x))=g(x)+k.
\tag{1}
\]
Let $a$ and $b$ be the least values attained by $f$ and $g$, respectively, and choose $n_f,n_g$ such that
\[
f(n_f)=a,
\qquad
g(n_g)=b.
\]
By (1), the images of the two functions are
\[
N_f=\{a,a+1,a+2,\ldots\},
\qquad
N_g=\{b,b+1,b+2,\ldots\}.
\tag{2}
\]

We have
\[
f(x)=f(y)\quad\Longleftrightarrow\quad g(x)=g(y).
\tag{3}
\]
Indeed, the forward implication follows from
\[
g(x)=g(f(x))-1=g(f(y))-1=g(y),
\]
and the converse is analogous. Define $x\sim y$ when these equivalent equalities hold, and let $[x]$ be the equivalence class of $x$.

\textbf{Claim 1.} If $f(x)\sim f(y)$, then $x\sim y$. Consequently, each equivalence class contains at most one element of $N_f$ and at most one element of $N_g$.

If $f(x)\sim f(y)$, then
\[
g(x)=g(f(x))-1=g(f(y))-1=g(y),
\]
so $x\sim y$. For the consequence, if two values $f(x),f(y)\in N_f$ lie in the same class, then Claim 1 gives $x\sim y$, hence $f(x)=f(y)$. The argument for $N_g$ is the same.

\textbf{Claim 2.} For every $x\in\mathbb{N}$,
\[
[x]\subseteq\{1,\ldots,b-1\}
\quad\Longleftrightarrow\quad
f(x)=a.
\tag{4}
\]
Similarly,
\[
[x]\subseteq\{1,\ldots,a-1\}
\quad\Longleftrightarrow\quad
g(x)=b.
\tag{5}
\]

We prove (4) by proving its complementary form. If $f(x)>a$, then by (2) there is $y$ with $f(y)=f(x)-1$. Hence
\[
f(g(y))=f(y)+1=f(x),
\]
so $x\sim g(y)$, and $g(y)\ge b$. Conversely, if $x\sim c$ for some $c\ge b$, then $c=g(y)$ for some $y$, and
\[
f(x)=f(c)=f(g(y))=f(y)+1\ge a+1.
\]
This proves (4), and (5) follows symmetrically.

Assume first that $a\le b$. By Claim 2, there is exactly one class contained in $\{1,\ldots,a-1\}$, namely $[n_g]$, and exactly one class contained in $\{1,\ldots,b-1\}$, namely $[n_f]$. Since the first set is contained in the second, these classes coincide. Thus
\[
f(x)=a
\quad\Longleftrightarrow\quad
g(x)=b
\quad\Longleftrightarrow\quad
x\sim n_f\sim n_g.
\tag{6}
\]

\textbf{Claim 3.} We have $a=b$.

First note that $f(a)\ne a$, for otherwise
\[
g(a)=g(f(a))=g(a)+1,
\]
a contradiction. Hence $[a]\ne[n_f]$. By Claim 2, the class $[a]$ contains some $a'\ge b$. Since both $a$ and $a'$ belong to $N_f$, Claim 1 forces $a=a'$. Therefore $a\ge b$. Together with $a\le b$, this gives $a=b$. The other possible initial ordering is handled symmetrically.

\textbf{Claim 4.} For every integer $d\ge 0$,
\[
f^{d+1}(n_f)=g^{d+1}(n_f)=a+d.
\tag{7}
\]
For $d=0$, this follows from (6) and Claim 3. Suppose it holds up to $d-1$. Then
\[
\begin{aligned}
f^{d+1}(n_f)
&=f(f^d(n_f))
=f(g^d(n_f))\\
&=f(n_f)+d=a+d,
\end{aligned}
\]
where the last equality follows from (1). The formula for $g^{d+1}(n_f)$ is analogous.

Finally, take any $x\in\mathbb{N}$. By (2), $f(x)=a+d$ for some $d\ge 0$. From (1) and (7),
\[
f(x)=f(g^d(n_f)),
\]
so $x\sim g^d(n_f)$. Therefore
\[
g(x)=g(g^d(n_f))=g^{d+1}(n_f)=a+d=f(x).
\]

SOLUTION 2:
Use the equivalence relation $x\sim y$ defined by
\[
f(x)=f(y)
\quad\Longleftrightarrow\quad
g(x)=g(y),
\]
and retain Claim 1 from Solution 1: if $f(x)\sim f(y)$, then $x\sim y$.

Let $a$ and $b$ be the minimum values of $f$ and $g$. We have $f(a)>a$; otherwise $f(a)=a$ and
\[
g(a)=g(f(a))=g(a)+1,
\]
which is impossible.

\textbf{Claim 1.} We have $a=b$.

Assume without loss of generality that $a\le b$. Since $f(a)\ge a+1$ and every integer at least $a$ lies in the image of $f$, there is $x$ such that
\[
f(a)=f(x)+1=f(g(x)).
\]
Thus $a\sim g(x)$. Since $g(x)\ge b\ge a$, Claim 1 from Solution 1 shows that the two image values $a$ and $g(x)$ in the same class must be equal. Hence
\[
a=g(x)\ge b.
\]
Together with $a\le b$, this gives $a=b$.

\textbf{Claim 2.} We have
\[
f(a)=g(a)=a+1.
\tag{1}
\]
Suppose $f(a)\ge a+2$. Choose $x,y\in\mathbb{N}$ such that
\[
f(x)=f(a)-2,
\qquad
f(y)=g(x),
\]
which is possible because $g(x)\ge a=b$ and all integers at least $a$ are values of $f$. Then
\[
f(a)=f(x)+2=f(g^2(x)),
\]
so $a\sim g^2(x)$. But $g^2(x)\ge a$, and Claim 1 from Solution 1 gives
\[
a=g^2(x)=g(f(y))=g(y)+1\ge a+1,
\]
a contradiction. Hence $f(a)=a+1$. The proof that $g(a)=a+1$ is symmetric.

\textbf{Claim 3.} For every integer $x\ge a$,
\[
f(x)=g(x)=x+1.
\tag{2}
\]
This follows by induction. The base case is (1). If (2) holds at $x$, then
\[
f(x+1)=f(g(x))=f(x)+1=x+2,
\]
and similarly
\[
g(x+1)=g(f(x))=g(x)+1=x+2.
\]

For arbitrary $n\in\mathbb{N}$, we have $g(n)\ge a$. Hence Claim 3 gives
\[
f(n)+1=f(g(n))=g(n)+1,
\]
and therefore $f(n)=g(n)$.
